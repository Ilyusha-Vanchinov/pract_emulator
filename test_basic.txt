 Базовый тест эмулятора
# Проверка вывода текущей директории
pwd

# Проверка корневой директории (должны увидеть bin/ и home/)
ls /

# Переход в папку пользователя
cd /home/user

# Проверка содержимого папки (должны увидеть docs/ и hello.txt)
ls

# Чтение файла hello.txt (должно вывести Hello World!)
cat hello.txt

# Чтение файла во вложенной папке
cat docs/report.txt

# Выход из эмулятора
exit
